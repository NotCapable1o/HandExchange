-- 1. DROP everything to start fresh
DROP VIEW IF EXISTS public.chat_list_view;

-- 2. Create the View with LEFT JOINS
-- This ensures that even if an image or title is missing, the chat STILL shows up.
CREATE VIEW public.chat_list_view AS
SELECT DISTINCT ON (cr.id)
    cr.id AS room_id,
    cr.buyer_id,
    cr.seller_id,
    cr.last_message_at,
    p.title AS last_product_title,
    p.images AS product_images,
    b.full_name AS buyer_name,
    b.avatar_url AS buyer_avatar,
    s.full_name AS seller_name,
    s.avatar_url AS seller_avatar
FROM chat_rooms cr
LEFT JOIN products p ON cr.product_id = p.id
LEFT JOIN profiles b ON cr.buyer_id = b.id
LEFT JOIN profiles s ON cr.seller_id = s.id
ORDER BY cr.id, cr.last_message_at DESC;

-- 3. Update Trigger (The Messenger "Jump to Top" Logic)
CREATE OR REPLACE FUNCTION update_room_timestamp()
RETURNS TRIGGER AS $$
BEGIN
  UPDATE chat_rooms
  SET last_message_at = NOW()
  WHERE id = NEW.room_id;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS tr_refresh_chat_order ON messages;
CREATE TRIGGER tr_refresh_chat_order
AFTER INSERT ON messages
FOR EACH ROW
EXECUTE FUNCTION update_room_timestamp();

-- 4. CRITICAL: Grant permission to access the view
GRANT SELECT ON public.chat_list_view TO authenticated;
GRANT SELECT ON public.chat_list_view TO anon;
