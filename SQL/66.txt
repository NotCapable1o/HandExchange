-- 1. Fix Deletion: This allows chat_rooms to be deleted without "Foreign Key" errors
ALTER TABLE public.messages
DROP CONSTRAINT IF EXISTS messages_room_id_fkey,
ADD CONSTRAINT messages_room_id_fkey
FOREIGN KEY (room_id) REFERENCES chat_rooms(id) ON DELETE CASCADE;

-- 2. Trigger Function: Updates the room timestamp when a new message arrives
CREATE OR REPLACE FUNCTION handle_new_message()
RETURNS TRIGGER AS $$
BEGIN
  UPDATE chat_rooms
  SET last_message_at = NOW()
  WHERE id = NEW.room_id;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 3. Attach Trigger: Ensures the Chat List stream sees the update immediately
DROP TRIGGER IF EXISTS on_message_sent ON messages;
CREATE TRIGGER on_message_sent
AFTER INSERT ON messages
FOR EACH ROW
EXECUTE FUNCTION handle_new_message();
