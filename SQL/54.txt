-- 1. Create the function that updates the timestamp
CREATE OR REPLACE FUNCTION update_room_timestamp()
RETURNS TRIGGER AS $$
BEGIN
  UPDATE chat_rooms
  SET last_message_at = NOW()
  WHERE id = NEW.room_id;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 2. Create the trigger to run that function automatically
CREATE TRIGGER tr_update_last_message
AFTER INSERT ON messages
FOR EACH ROW
EXECUTE FUNCTION update_room_timestamp();
