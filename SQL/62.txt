-- This function updates the room's timestamp
CREATE OR REPLACE FUNCTION bump_chat_to_top()
RETURNS TRIGGER AS $$
BEGIN
  UPDATE chat_rooms
  SET last_message_at = NOW()
  WHERE id = NEW.room_id;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- This trigger runs the function automatically on every new message
DROP TRIGGER IF EXISTS tr_bump_chat ON messages;
CREATE TRIGGER tr_bump_chat
AFTER INSERT ON messages
FOR EACH ROW
EXECUTE FUNCTION bump_chat_to_top();
