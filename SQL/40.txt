
-- 1. Make the bucket public (if not already)
-- 2. Allow users to upload their own folder
CREATE POLICY "Avatar Upload Policy" ON storage.objects
FOR INSERT WITH CHECK (bucket_id = 'avatars' AND auth.uid()::text = (storage.foldername(name))[1]);

-- 3. Allow users to update/overwrite their own avatar
CREATE POLICY "Avatar Update Policy" ON storage.objects
FOR UPDATE USING (bucket_id = 'avatars' AND auth.uid()::text = (storage.foldername(name))[1]);

-- 4. Allow public to view avatars
CREATE POLICY "Public Avatar Access" ON storage.objects
FOR SELECT USING (bucket_id = 'avatars');
