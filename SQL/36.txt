
-- 1. Create a policy to allow anyone to view profile images
CREATE POLICY "Public Profile Images" ON storage.objects FOR SELECT USING (bucket_id = 'avatars');

-- 2. Allow users to upload their own profile image
CREATE POLICY "Users can upload own avatar" ON storage.objects FOR INSERT WITH CHECK (
  bucket_id = 'avatars' AND auth.uid() = owner
);

-- 3. Allow users to update/delete their own avatar
CREATE POLICY "Users can update own avatar" ON storage.objects FOR UPDATE USING (bucket_id = 'avatars' AND auth.uid() = owner);
CREATE POLICY "Users can delete own avatar" ON storage.objects FOR DELETE USING (bucket_id = 'avatars' AND auth.uid() = owner);
