-- 1. Add missing columns to profiles and messages
ALTER TABLE public.profiles ADD COLUMN IF NOT EXISTS show_active_status BOOLEAN DEFAULT true;
-- Use BIGINT for reply_to because the 'id' column in messages is likely bigint
ALTER TABLE public.messages ADD COLUMN IF NOT EXISTS reply_to BIGINT REFERENCES messages(id);
ALTER TABLE public.messages ADD COLUMN IF NOT EXISTS reactions JSONB DEFAULT '[]';
ALTER TABLE public.messages ADD COLUMN IF NOT EXISTS is_read BOOLEAN DEFAULT false;

-- 2. Fix Deletion: Allow chat rooms to be deleted even if they contain messages
ALTER TABLE public.messages
DROP CONSTRAINT IF EXISTS messages_room_id_fkey,
ADD CONSTRAINT messages_room_id_fkey
FOREIGN KEY (room_id) REFERENCES chat_rooms(id) ON DELETE CASCADE;

-- 3. Trigger: Automatically update the chat room timestamp when a message is sent
CREATE OR REPLACE FUNCTION handle_new_message()
RETURNS TRIGGER AS $$
BEGIN
  UPDATE chat_rooms SET last_message_at = NOW() WHERE id = NEW.room_id;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS on_message_sent ON messages;
CREATE TRIGGER on_message_sent AFTER INSERT ON messages FOR EACH ROW EXECUTE FUNCTION handle_new_message();

-- 4. Recreate the Messenger View with correct unread logic and active status
DROP VIEW IF EXISTS public.chat_list_view;
CREATE VIEW public.chat_list_view AS
SELECT DISTINCT ON (cr.id)
    cr.id AS room_id,
    cr.buyer_id, cr.seller_id, cr.last_message_at,
    p.title AS last_product_title,
    b.full_name AS buyer_name, b.avatar_url AS buyer_avatar, b.show_active_status AS buyer_active_enabled,
    s.full_name AS seller_name, s.avatar_url AS seller_avatar, s.show_active_status AS seller_active_enabled,
    -- Count messages NOT sent by the current user
    (SELECT count(*)::int FROM messages WHERE room_id = cr.id AND is_read = false AND sender_id != auth.uid()) AS unread_count,
    (SELECT content FROM messages WHERE room_id = cr.id ORDER BY created_at DESC LIMIT 1) AS last_message_snippet
FROM chat_rooms cr
LEFT JOIN products p ON cr.product_id = p.id
LEFT JOIN profiles b ON cr.buyer_id = b.id
LEFT JOIN profiles s ON cr.seller_id = s.id
ORDER BY cr.id, cr.last_message_at DESC;
